scn ExtProbeStrings01
; Sv_Length, Sv_Find, Sv_Count (tools/capture/ext-probes/strings/README.md)
string_var sS
string_var sT
array_var aA
int iI
float fF
ref rRef
int iDone
Begin GameMode
    if iDone
        Return
    endif
    let iDone := 1
    Print "EXTPROBE|strings/01_length_find|begin"
    let sS := "Ringo"
    Print "EXTPROBE|len.basic|" + $(Sv_Length sS)
    let sS := ""
    Print "EXTPROBE|len.empty|" + $(Sv_Length sS)
    let sS := "x"
    Sv_Destruct sS
    Print "EXTPROBE|len.destructed|" + $(Sv_Length sS)
    let sS := "Hello there"
    Print "EXTPROBE|find.first|" + $(Sv_Find "e", sS)
    let sS := "Hello there"
    Print "EXTPROBE|find.nocase|" + $(Sv_Find "hello", sS)
    let sS := "Hello there"
    Print "EXTPROBE|find.case_flag|" + $(Sv_Find "hello", sS, 0, 11, 1)
    let sS := "Hello there"
    Print "EXTPROBE|find.missing|" + $(Sv_Find "xyz", sS)
    let sS := "Hello there"
    Print "EXTPROBE|find.range|" + $(Sv_Find "e", sS, 5, 6)
    let sS := "Hello there"
    Print "EXTPROBE|count.basic|" + $(Sv_Count "l", sS)
    let sS := "the theme"
    Print "EXTPROBE|count.in_word|" + $(Sv_Count "the", sS)
    let sS := "aaaa"
    Print "EXTPROBE|count.overlap|" + $(Sv_Count "aa", sS)
    Print "EXTPROBE|strings/01_length_find|end"
End
