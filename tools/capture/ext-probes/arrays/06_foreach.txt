    let iR := 0
    foreach aE <- aA
        let iR := iR * 10 + aE["key"]
    loop
    Print "EXTPROBE|foreach.keys|" + $(iR)
    let aA := Ar_Null
    let sS := "abc"

        if eval *aE == 3
            break

    loop
    Print "EXTPROBE|foreach.break|" + $(iR)

    let iR := 0

    loop

    let aA := Ar_Null
    let aA := Ar_List 1, 2
    let iR := 0

        endif
    loop
