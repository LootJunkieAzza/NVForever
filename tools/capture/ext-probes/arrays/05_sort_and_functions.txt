; Ar_Sort, Ar_SortAlpha, Ar_CustomSort, Ar_Filter, Ar_MapTo, Ar_FindWhere, ar_Any/All, Ar_ForEach (tools/capture/ext-probes/arrays/README.md)
array_var aA

Begin GameMode
    if iDone
        Return
    endif
    let iDone := 1
    Print "EXTPROBE|arrays/05_sort_and_functions|begin"
    let aA := Ar_Null
    let aA := Ar_List 3, 1, 2
    let aB := Ar_Sort aA
    Print "EXTPROBE|sort.asc|" + $(aB[0])
    let aA := Ar_Null
    let aA := Ar_List 3, 1, 2
    let aB := Ar_Sort aA, 1
    Print "EXTPROBE|sort.desc|" + $(aB[0])
    let aA := Ar_Null
    let aA := Ar_List 3, "x"
    let aB := Ar_Sort aA
    Print "EXTPROBE|sort.mixed_empty|" + $(Ar_Size aB)
    let aA := Ar_Null
    let aA := Ar_List "b", "A", "c"
    let aB := Ar_Sort aA
    Print "EXTPROBE|sort.strings_nocase|[" + (aB[0]) + "]"
    let aA := Ar_Null
    let aA := Ar_Map 1::30, 2::10
    let aB := Ar_Sort aA
    Print "EXTPROBE|sort.map_to_list|[" + (TypeOf aB) + "]"
    let aA := Ar_Null
    let aA := Ar_List 10, 9
    let aB := Ar_SortAlpha aA
    Print "EXTPROBE|sortalpha.numbers_as_text|" + $(aB[0])
    let aA := Ar_Null
    let aA := Ar_List 1, 20, 3, 40
    let aB := Ar_Filter aA, ({aX} => *aX > 10)
    Print "EXTPROBE|filter.size|" + $(Ar_Size aB)
    let aA := Ar_Null
    let aA := Ar_List 1, 4
    let aB := Ar_MapTo aA, ({aX} => *aX * 2)
    Print "EXTPROBE|mapto|" + $(aB[1])
    let aA := Ar_Null
    let aA := Ar_List 5, 12, 8
    let iR := Ar_FindWhere aA, ({aX} => *aX > 10)
    Print "EXTPROBE|findwhere|" + $(iR)
    let aA := Ar_Null
    let aA := Ar_List 5, 8
    let iR := Ar_FindWhere aA, ({aX} => *aX > 10)
    Print "EXTPROBE|findwhere.none|" + $(iR)
    let aA := Ar_Null
    let aA := Ar_List 1, 50
    Print "EXTPROBE|any.yes|" + $(ar_Any aA, ({aX} => *aX > 10))
    let aA := Ar_Null
    let aA := Ar_List 1, 50
    Print "EXTPROBE|all.no|" + $(ar_All aA, ({aX} => *aX > 10))
    let aA := Ar_Null
    let aA := Ar_List 1, 2, 3
    let iR := 0
    Ar_ForEach aA, ({aX} => iR += *aX)
    Print "EXTPROBE|foreach.sum|" + $(iR)
    Print "EXTPROBE|arrays/05_sort_and_functions|end"
End
