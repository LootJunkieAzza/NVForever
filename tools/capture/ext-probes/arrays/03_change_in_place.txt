    Print "EXTPROBE|append|" + $(aA[1])

    Ar_Append aA, 2
    Print "EXTPROBE|append.map|" + $(Ar_Size aA)
    let aA := Ar_Null
    let aA := Ar_List 1, 3
    Ar_Insert aA, 1, 2
    Print "EXTPROBE|insert.middle|" + $(aA[1])
    let aA := Ar_Null

    let iR := Ar_Insert aA, 2, 3
    Print "EXTPROBE|insert.at_size|" + $(iR)
    let aA := Ar_Null
    let aA := Ar_List 1, 2
    let iR := Ar_Insert aA, 5, 3
    Print "EXTPROBE|insert.past|" + $(iR)
    let aA := Ar_Null
    let aA := Ar_List 1, 5
    let aB := Ar_List 2, 3, 4
    Ar_InsertRange aA, 1, aB

    let aA := Ar_List 1, 2, 3, 4

    let aA := Ar_List 1
    Ar_Resize aA, 3
    Print "EXTPROBE|resize.pad_default|" + $(aA[2])
    let aA := Ar_Null
