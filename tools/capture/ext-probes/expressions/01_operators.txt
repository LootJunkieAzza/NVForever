scn ExtProbeExpr01
; operators, associativity, && and ||, unary, bitwise (tools/capture/ext-probes/expressions/README.md)
int iX
int iY
int iR
float fF
int iDone
Begin GameMode
    if iDone
        Return
    endif
    let iDone := 1
    Print "EXTPROBE|expressions/ExtProbeExpr01|begin"
    Print "EXTPROBE|assoc.pow|" + $(2 ^ 3 ^ 2)
    Print "EXTPROBE|assoc.sub|" + $(10 - 4 - 3)
    Print "EXTPROBE|assoc.div|" + $(64 / 4 / 2)
    Print "EXTPROBE|assoc.mod|" + $(17 % 5 % 3)
    let iX := 0
    Print "EXTPROBE|assign.value|" + $((iX := 4) + 1)
    let iX := 0
    let iY := 0
    let iX := iY := 7
    Print "EXTPROBE|assign.chain_a|" + $(iX)
    let iX := 0
    let iY := 0
    let iX := iY := 7
    Print "EXTPROBE|assign.chain_b|" + $(iY)
    let iR := 0 || 5
    Print "EXTPROBE|or.value|" + $(iR)
    let iR := 1 || 2
    Print "EXTPROBE|or.left|" + $(iR)
    let iR := 0 || 0
    Print "EXTPROBE|or.zero|" + $(iR)
    let iR := 3 && 4
    Print "EXTPROBE|and.value|" + $(iR)
    let iR := 3 && 0
    Print "EXTPROBE|and.zero_right|" + $(iR)
    let iR := 0 && 5
    Print "EXTPROBE|and.zero_left|" + $(iR)
    let iR := 1 && -5
    Print "EXTPROBE|and.negative|" + $(iR)
    let iR := 0 || 2 && 3
    Print "EXTPROBE|andor.prec|" + $(iR)
    let iR := 1 || 0 && 0
    Print "EXTPROBE|andor.prec2|" + $(iR)
    Print "EXTPROBE|not.plus|" + $(!0 + 1)
    Print "EXTPROBE|neg.pow|" + $(-2 ^ 2)
    Print "EXTPROBE|shift.left|" + $(1 << 3)
    Print "EXTPROBE|shift.right_neg|" + $(-8 >> 1)
    Print "EXTPROBE|bitor.fraction|" + $(5.7 | 0)
    Print "EXTPROBE|bitor.neg_fraction|" + $(-5.7 | 0)
    Print "EXTPROBE|bitand.fraction|" + $(5.9 & 7)
    Print "EXTPROBE|mod.fraction|" + $(7.5 % 2)
    Print "EXTPROBE|paren.fn|" + $((Sin 90) > 0.5)
    let fF := Rand 5, 5
    Print "EXTPROBE|rand.comma|" + $(fF)
    Print "EXTPROBE|expressions/ExtProbeExpr01|end"
End
