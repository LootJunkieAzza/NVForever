; foreach over a list, a string map, a string, a form list and a container (tools/capture/ext-probes/loops/README.md)

int j

int iFrame
ref rItem

array_var a
array_var e

string_var c
Begin GameMode
    let iFrame += 1
    if iFrame == 1
        Print "EXTPROBE|loops/ExtProbeLoops02|begin"
        let a := Ar_List 10, 20, 30
        let r := 0
        foreach e <- a
            let r := r * 10 + e["key"]
        loop
        Print "EXTPROBE|foreach.list_keys|" + $r
        let r := 0
        foreach e <- a
            let r += *e
        loop
        Print "EXTPROBE|foreach.list_values|" + $r
        let a := Ar_Map "b"::2, "a"::1, "c"::3
        let s := ""
        foreach e <- a
            let s += e["key"]
        loop
        Print "EXTPROBE|foreach.smap_order|[" + s + "]"
        let s := "xyz"
        let r := 0
        foreach c <- s
            let r += 1
        loop
        Print "EXTPROBE|foreach.string_count|" + $r
        let s := "xyz"
        let a := Ar_Construct "array"
        foreach c <- s
            Ar_Append a, c
        loop
        Print "EXTPROBE|foreach.string_last|[" + a[2] + "]"
        let r := 0
        foreach rForm <- ExtProbeList
            let r += 1
        loop
        Print "EXTPROBE|foreach.formlist_count|" + $r
        Print "EXTPROBE|foreach.formlist_listcount|" + $(ListGetCount ExtProbeList)
        let r := 0
        foreach rItem <- PlayerREF
            let r += 1
            let rKeep := rItem
        loop
        Print "EXTPROBE|foreach.container_entries|" + $r
        Print "EXTPROBE|foreach.container_kinds|" + $(PlayerREF.GetNumItems)
        Print "EXTPROBE|foreach.tempref_after_loop|" + $(IsFormValid rKeep)
    elseif iFrame == 2
        Print "EXTPROBE|foreach.tempref_next_frame|" + $(IsFormValid rKeep)
        Print "EXTPROBE|loops/ExtProbeLoops02|end"
    endif
End
