Begin GameMode
    if iDone

    endif
    let iDone := 1
    Print "EXTPROBE|udf/ExtProbeUdf01|begin"
    Print "EXTPROBE|udf.add|" + $(Call ExtUdfAdd 2 3)
    let i := 5
    let r := Call ExtUdfSetNum i
    Print "EXTPROBE|udf.num_by_value|" + $i
    let s := "hi"
    let r := Call ExtUdfSetStr s
    Print "EXTPROBE|udf.str_after_call|[" + s + "]"

    let r := Call ExtUdfSetArr a
    Print "EXTPROBE|udf.arr_after_call|" + $(a[0])
    Print "EXTPROBE|udf.twice|" + $(Call ExtUdfTwice)
    let r := Call ExtUdfAfterValue
    let r := 42
    let r := Call ExtUdfNoValue
    Print "EXTPROBE|udf.no_value|" + $r
    Print "EXTPROBE|udf.return_stops|" + $(Call ExtUdfReturnStops)
    Print "EXTPROBE|udf/ExtProbeUdf01|end"
